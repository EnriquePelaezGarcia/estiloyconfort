# PLAN DE DESARROLLO MODULAR
## Mueblería Estilo y Confort - Aplicación Web

He analizado tu documentación completa, las imágenes de diseño, los archivos HTML/CSS y los recursos gráficos. Diseño un plan de desarrollo en **5 fases modulares** que permite liberar software funcional al final de cada etapa, asegurar consistencia visual y facilitar ajustes iterativos.

---

## 📋 ESTRUCTURA DE FASES

```
FASE 1: Infraestructura Base + Autenticación (2-3 semanas)
    ↓
FASE 2: Catálogo Público + Carrito Anónimo (2-3 semanas)
    ↓
FASE 3: Dashboard Administrador (2 semanas)
    ↓
FASE 4: Módulos de Negocio (Vendedor, Repartidor, Fabricante) (2-3 semanas)
    ↓
FASE 5: Optimización, Testing y Deployment (1-2 semanas)
```

#### Base de Datos:
```sql
CREATE TABLE categories (
  id INT PRIMARY KEY AUTO_INCREMENT,
  name VARCHAR(255) UNIQUE,
  description TEXT,
  image_url VARCHAR(255),
  order_display INT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE products (
  id INT PRIMARY KEY AUTO_INCREMENT,
  name VARCHAR(255) NOT NULL,
  sku VARCHAR(100) UNIQUE,
  category_id INT,
  description TEXT,
  materials VARCHAR(255),
  dimensions_length DECIMAL(8,2),    // cm
  dimensions_width DECIMAL(8,2),     // cm
  dimensions_height DECIMAL(8,2),    // cm
  weight_volumetric DECIMAL(8,2),    // kg
  availability_days INT,              // 0 = inmediata
  base_cost DECIMAL(12,2),
  margin_percentage DECIMAL(5,2),
  price_base_no_iva DECIMAL(12,2),
  price_with_iva DECIMAL(12,2),
  price_cash_absorbing_fee DECIMAL(12,2),
  price_6msi_absorbing_fee DECIMAL(12,2),
  stock_quantity INT DEFAULT 0,
  stock_alert_level INT DEFAULT 5,
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (category_id) REFERENCES categories(id)
);

CREATE TABLE product_variants (
  id INT PRIMARY KEY AUTO_INCREMENT,
  product_id INT,
  variant_type VARCHAR(100),        // "color", "tamaño", "tapiz"
  variant_value VARCHAR(100),
  price_modifier DECIMAL(12,2),     // Suma o resta al precio base
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (product_id) REFERENCES products(id)
);

CREATE TABLE product_images (
  id INT PRIMARY KEY AUTO_INCREMENT,
  product_id INT,
  image_url VARCHAR(255),
  alt_text VARCHAR(255),
  is_primary BOOLEAN DEFAULT FALSE,
  order_display INT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (product_id) REFERENCES products(id)
);
```

CREATE TABLE orders (
  id INT PRIMARY KEY AUTO_INCREMENT,
  order_number VARCHAR(20) UNIQUE,
  seller_id INT,
  customer_name VARCHAR(255),
  customer_email VARCHAR(255),
  customer_phone VARCHAR(20),
  delivery_address TEXT,
  delivery_address_lat DECIMAL(10,8),
  delivery_address_lng DECIMAL(11,8),
  delivery_type ENUM('standard', 'with_installation'),
  delivery_person_id INT,
  payment_method ENUM('cash', 'card', 'msi', 'store_credit'),
  payment_status ENUM('pending', 'partial', 'paid'),
  payment_amount DECIMAL(12,2),
  order_status ENUM('pending', 'fabricating', 'ready', 'in_delivery', 'delivered', 'cancelled'),
  order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  expected_delivery_date DATE,
  total_amount DECIMAL(12,2),
  notes TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (seller_id) REFERENCES users(id),
  FOREIGN KEY (delivery_person_id) REFERENCES users(id)
);

CREATE TABLE order_items (
  id INT PRIMARY KEY AUTO_INCREMENT,
  order_id INT,
  product_id INT,
  quantity INT,
  variant_selections JSON,  // {"color": "negro", "size": "2 plazas"}
  unit_price DECIMAL(12,2),
  subtotal DECIMAL(12,2),
  FOREIGN KEY (order_id) REFERENCES orders(id),
  FOREIGN KEY (product_id) REFERENCES products(id)
);

CREATE TABLE deliveries (
  id INT PRIMARY KEY AUTO_INCREMENT,
  order_id INT UNIQUE,
  delivery_person_id INT,
  assignment_date DATE,
  delivery_status ENUM('pending', 'in_progress', 'completed', 'failed'),
  signature_image_url VARCHAR(255),
  photo_url VARCHAR(255),
  delivered_at TIMESTAMP,
  notes TEXT,
  FOREIGN KEY (order_id) REFERENCES orders(id),
  FOREIGN KEY (delivery_person_id) REFERENCES users(id)
);

CREATE TABLE payments (
  id INT PRIMARY KEY AUTO_INCREMENT,
  order_id INT,
  amount DECIMAL(12,2),
  payment_method ENUM('cash', 'card', 'msi', 'store_credit'),
  payment_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  collected_by_id INT,
  FOREIGN KEY (order_id) REFERENCES orders(id),
  FOREIGN KEY (collected_by_id) REFERENCES users(id)
);
